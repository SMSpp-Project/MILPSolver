# PIPS-only makefile fragment for MILPSolver.
#
# This builds only MILPSolver.o and PIPSMILPSolver.o, and generates only the
# PIPS parameter headers. It does not touch CPLEX/GUROBI/SCIP/HiGHS.

TOOLSSDR := ./$(MILPSSDR)/tools
STAMP := $(TOOLSSDR)/.headers-pips.stamp

# PIPS-IPM++ source tree, needed only by tools/pips_pars.
# Override from the command line if your layout differs, e.g.
#   make PIPSIPM_ROOT=/absolute/path/to/pips-ipmpp
PIPSIPM_ROOT ?= $(PIPS_ROOT)

.PHONY: tools-pips clean-pips distclean-pips

tools-pips: $(STAMP)

$(STAMP):
	@echo "[MILPSolver] building and running pips_pars in $(TOOLSSDR)"
	@set -e; \
	$(MAKE) -C "$(TOOLSSDR)" pips_pars; \
	cd "$(TOOLSSDR)"; \
	if find ../include -maxdepth 1 -name "PIPS*_defs.h" -print -quit | grep -q . && \
	   find ../include -maxdepth 1 -name "PIPS*_maps.h" -print -quit | grep -q . ; then \
	  echo " -> pips_pars (skip: headers already present)"; \
	else \
	  echo " -> pips_pars"; \
	  ./pips_pars -s "$(PIPSIPM_ROOT)" ../include; \
	fi
	@echo "[MILPSolver] PIPS headers ready"
	@touch "$(STAMP)"

MILPSOBJ = $(MILPSSDR)/obj/MILPSolver.o \
	$(MILPSSDR)/obj/PIPSMILPSolver.o

MILPSINC = -I$(MILPSSDR)/include

MILPSH = $(MILPSSDR)/include/MILPSolver.h \
	$(MILPSSDR)/include/PIPSMILPSolver.h

clean-pips:
	rm -f $(MILPSSDR)/obj/MILPSolver.o
	rm -f $(MILPSSDR)/obj/PIPSMILPSolver.o

distclean-pips: clean-pips
	rm -f $(MILPSSDR)/include/PIPS*_defs.h
	rm -f $(MILPSSDR)/include/PIPS*_maps.h
	rm -f $(STAMP)

$(MILPSSDR)/obj/MILPSolver.o: $(MILPSSDR)/src/MILPSolver.cpp \
	$(MILPSSDR)/include/MILPSolver.h $(SMS++OBJ)
	$(CC) -c $(MILPSSDR)/src/MILPSolver.cpp -o $@ \
	-I$(MILPSSDR)/include $(SMS++INC) $(SW)

$(MILPSSDR)/obj/PIPSMILPSolver.o: $(MILPSSDR)/src/PIPSMILPSolver.cpp \
	$(MILPSSDR)/include/PIPSMILPSolver.h \
	$(MILPSSDR)/include/MILPSolver.h $(SMS++OBJ) | $(STAMP)
	$(CC) -c $(MILPSSDR)/src/PIPSMILPSolver.cpp -o $@ \
	-I$(MILPSSDR)/include $(SMS++INC) $(libPIPSINC) $(SW)

########################### End of makefile-pips #############################
